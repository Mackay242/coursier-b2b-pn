   261→function NotificationPanel({ open, onClose }: { open: boolean; onClose: () => void }) {
   262→  const [notifications, setNotifications] = useState<AppNotification[]>([]);
   263→  const [unreadCount, setUnreadCount] = useState(0);
   264→  const [loading, setLoading] = useState(true);
   265→
   266→  const fetchNotifs = useCallback(async () => {
   267→    try {
   268→      const res = await fetch('/api/notifications');
   269→      if (res.ok) {
   270→        const data = await res.json();
   271→        setNotifications(data.notifications || []);
   272→        setUnreadCount(data.nonLues || 0);
   273→      }
   274→    } catch { /* ignore */ } finally { setLoading(false); }
   275→  }, []);
   276→
   277→  useEffect(() => {
   278→    if (open) fetchNotifs();
   279→  }, [open, fetchNotifs]);
   280→
   281→  const markAsRead = async (id: string) => {
   282→    await fetch(`/api/notifications/${id}`, { method: 'PATCH' });
   283→    setNotifications(prev => prev.map(n => n.id === id ? { ...n, read: true } : n));
   284→    setUnreadCount(prev => Math.max(0, prev - 1));
   285→  };
   286→
   287→  const markAllRead = async () => {
   288→    await fetch(`/api/notifications/all`, { method: 'PATCH' });
   289→    setNotifications(prev => prev.map(n => ({ ...n, read: true })));
   290→    setUnreadCount(0);
   291→  };
   292→
   293→  const notifIcons: Record<string, React.ReactNode> = {
   294→    new_delivery: <Package className="w-4 h-4" />,
   295→    status_change: <Navigation className="w-4 h-4" />,
   296→    payment: <CircleDollarSign className="w-4 h-4" />,
   297→    system: <Bell className="w-4 h-4" />,
   298→  };
   299→
   300→  const timeAgo = (d: string) => {
   301→    const diff = Date.now() - new Date(d).getTime();
   302→    const mins = Math.floor(diff / 60000);
   303→    if (mins < 1) return "A l'instant";
   304→    if (mins < 60) return `Il y a ${mins} min`;
   305→    const hrs = Math.floor(mins / 60);
   306→    if (hrs < 24) return `Il y a ${hrs}h`;
   307→    return `Il y a ${Math.floor(hrs / 24)}j`;
   308→  };
   309→
   310→  return (
   311→    <>
   312→      {open && <div className="fixed inset-0 z-40" onClick={onClose} />}
   313→      <div className={`absolute right-0 top-full mt-2 w-96 max-h-[80vh] bg-card border rounded-xl shadow-2xl z-50 flex flex-col transition-all ${open ? 'opacity-100 translate-y-0' : 'opacity-0 -translate-y-2 pointer-events-none'}`}>
   314→        <div className="flex items-center justify-between p-4 border-b">
   315→          <h3 className="font-semibold text-sm">Notifications</h3>
   316→          <div className="flex items-center gap-2">
   317→            {unreadCount > 0 && <span className="text-[10px] text-muted-foreground">{unreadCount} non lue(s)</span>}
   318→            <Button variant="ghost" size="sm" className="h-7 text-xs" onClick={markAllRead} disabled={unreadCount === 0}>Tout marquer lu</Button>
   319→          </div>
   320→        </div>
   321→        <div className="flex-1 overflow-y-auto">
   322→          {loading ? (
   323→            <div className="p-4 space-y-3">{Array.from({ length: 4 }).map((_, i) => <div key={i} className="flex gap-3"><Skeleton className="w-8 h-8 rounded-full" /><div className="flex-1 space-y-1"><Skeleton className="h-3 w-3/4" /><Skeleton className="h-2.5 w-1/2" /></div></div>)}</div>
   324→          ) : notifications.length === 0 ? (
   325→            <div className="p-8 text-center text-muted-foreground">
   326→              <Bell className="w-8 h-8 mx-auto mb-2 opacity-30" />
   327→              <p className="text-sm">Aucune notification</p>
   328→            </div>
   329→          ) : (
   330→            notifications.map(n => (
   331→              <div key={n.id} className={`flex items-start gap-3 p-4 hover:bg-muted/50 cursor-pointer transition-colors border-b last:border-0 ${!n.read ? 'bg-primary/5' : ''}`} onClick={() => { if (!n.read) markAsRead(n.id); }}>
   332→                <div className={`w-8 h-8 rounded-full flex items-center justify-center shrink-0 ${!n.read ? 'bg-primary/10 text-primary' : 'bg-muted text-muted-foreground'}`}>{notifIcons[n.type] || <Bell className="w-4 h-4" />}</div>
   333→                <div className="flex-1 min-w-0">
   334→                  <div className="flex items-center gap-2">
   335→                    <p className={`text-sm ${!n.read ? 'font-semibold' : 'font-medium'}`}>{n.title}</p>
   336→                    {!n.read && <div className="w-2 h-2 rounded-full bg-primary shrink-0" />}
   337→                  </div>
   338→                  <p className="text-xs text-muted-foreground mt-0.5 line-clamp-2">{n.message}</p>
   339→                  <p className="text-[10px] text-muted-foreground mt-1">{timeAgo(n.createdAt)}</p>
   340→                </div>
   341→              </div>
   342→            ))
   343→          )}
   344→        </div>
   345→      </div>
   346→    </>
   347→  );
   348→}
