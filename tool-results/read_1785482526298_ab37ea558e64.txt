   229→function AuthForm() {
   230→  const [isRegister, setIsRegister] = useState(false);
   231→  const [email, setEmail] = useState('');
   232→  const [password, setPassword] = useState('');
   233→  const [name, setName] = useState('');
   234→  const [companyName, setCompanyName] = useState('');
   235→  const [phone, setPhone] = useState('');
   236→  const [plan, setPlan] = useState('business');
   237→  const [error, setError] = useState('');
   238→  const [loading, setLoading] = useState(false);
   239→
   240→  const handleLogin = async (e: React.FormEvent) => {
   241→    e.preventDefault();
   242→    setError('');
   243→    setLoading(true);
   244→    try {
   245→      const result = await signIn('credentials', { email, password, redirect: false });
   246→      if (result?.error) {
   247→        setError('Email ou mot de passe incorrect');
   248→      }
   249→    } catch {
   250→      setError('Erreur de connexion');
   251→    } finally {
   252→      setLoading(false);
   253→    }
   254→  };
   255→
   256→  const handleRegister = async (e: React.FormEvent) => {
   257→    e.preventDefault();
   258→    setError('');
   259→    setLoading(true);
   260→    try {
   261→      const res = await fetch('/api/auth/register', {
   262→        method: 'POST',
   263→        headers: { 'Content-Type': 'application/json' },
   264→        body: JSON.stringify({ email, password, name, companyName, phone, plan }),
   265→      });
   266→      const data = await res.json();
   267→      if (!res.ok) {
   268→        setError(data.error || "Erreur lors de l'inscription");
   269→        return;
   270→      }
   271→      // Auto-login after register
   272→      await signIn('credentials', { email, password, redirect: false });
   273→    } catch {
   274→      setError("Erreur serveur lors de l'inscription");
   275→    } finally {
   276→      setLoading(false);
   277→    }
   278→  };
   279→
   280→  return (
   281→    <div className="min-h-screen flex items-center justify-center p-4 relative overflow-hidden">
   282→      {/* Background */}
   283→      <div className="absolute inset-0 bg-gradient-to-br from-primary via-emerald-700 to-teal-800" />
   284→      <div className="absolute inset-0 opacity-10">
   285→        <div className="absolute top-20 left-20 w-64 h-64 rounded-full bg-white/20 blur-3xl" />

   289→
   290→      {/* Decorative elements */}
   291→      <div className="absolute top-8 left-8 opacity-20">

   293→          <circle cx="40" cy="40" r="35" stroke="white" strokeWidth="2" strokeDasharray="8 4" />
   294→          <path d="M25 40 L35 30 L35 36 L55 36 L55 44 L35 44 L35 50 Z" fill="white" fillOpacity="0.5" />
   295→        </svg>
   296→      </div>
   297→      <div className="absolute bottom-12 right-12 opacity-15">
   298→        <svg width="60" height="60" viewBox="0 0 60 60" fill="none" xmlns="http://www.w3.org/2000/svg">
   299→          <rect x="5" y="5" width="50" height="50" rx="8" stroke="white" strokeWidth="2" strokeDasharray="6 3" />
   300→          <circle cx="30" cy="30" r="12" stroke="white" strokeWidth="2" />
   301→        </svg>
   302→      </div>
   303→
   304→      <Card className="relative z-10 w-full max-w-md shadow-2xl border-0">
   305→        <CardContent className="p-0">
   306→          {/* Header */}
   307→          <div className="bg-gradient-to-r from-primary to-emerald-700 p-6 rounded-t-xl text-white text-center">
   308→            <div className="w-14 h-14 rounded-2xl bg-white/20 backdrop-blur flex items-center justify-center mx-auto mb-4">

   317→            <div className="flex bg-muted rounded-lg p-1 mb-6">
