   260→function NotificationPanel({ open, onClose }: { open: boolean; onClose: () => void }) {
   261→  const [notifications, setNotifications] = useState<AppNotification[]>([]);
   262→  const [unreadCount, setUnreadCount] = useState(0);
   263→  const [loading, setLoading] = useState(true);
   264→
   265→  const fetchNotifs = useCallback(async () => {
   266→    try {
   267→      const res = await fetch('/api/notifications');
   268→      if (res.ok) {
   269→        const data = await res.json();
   270→        setNotifications(data.notifications || []);
   271→        setUnreadCount(data.nonLues || 0);
   272→      }
   273→    } catch { /* ignore */ } finally { setLoading(false); }
   274→  }, []);
   275→
   276→  useEffect(() => {
   277→    if (open) fetchNotifs();
   278→  }, [open, fetchNotifs]);
   279→
   280→  const markAsRead = async (id: string) => {
   281→    await fetch(`/api/notifications/${id}`, { method: 'PATCH' });
   282→    setNotifications(prev => prev.map(n => n.id === id ? { ...n, read: true } : n));
   283→    setUnreadCount(prev => Math.max(0, prev - 1));
   284→  };
   285→
   286→  const markAllRead = async () => {
   287→    await fetch(`/api/notifications/all`, { method: 'PATCH' });
   288→    setNotifications(prev => prev.map(n => ({ ...n, read: true })));
   289→    setUnreadCount(0);
   290→  };
   291→
   292→  const notifIcons: Record<string, React.ReactNode> = {
   293→    new_delivery: <Package className="w-4 h-4" />,
   294→    status_change: <Navigation className="w-4 h-4" />,
   295→    payment: <CircleDollarSign className="w-4 h-4" />,
   296→    system: <Bell className="w-4 h-4" />,
   297→  };
   298→
   299→  const timeAgo = (d: string) => {
   300→    const diff = Date.now() - new Date(d).getTime();
   301→    const mins = Math.floor(diff / 60000);
   302→    if (mins < 1) return "A l'instant";
   303→    if (mins < 60) return `Il y a ${mins} min`;
   304→    const hrs = Math.floor(mins / 60);
   305→    if (hrs < 24) return `Il y a ${hrs}h`;
   306→    return `Il y a ${Math.floor(hrs / 24)}j`;
   307→  };
   308→
   309→  return (
   310→    <>
   311→      {open && <div className="fixed inset-0 z-40" onClick={onClose} />}
   312→      <div className={`absolute right-0 top-full mt-2 w-96 max-h-[80vh] bg-card border rounded-xl shadow-2xl z-50 flex flex-col transition-all ${open ? 'opacity-100 translate-y-0' : 'opacity-0 -translate-y-2 pointer-events-none'}`}>
   313→        <div className="flex items-center justify-between p-4 border-b">
   314→          <h3 className="font-semibold text-sm">Notifications</h3>
   315→          <div className="flex items-center gap-2">
   316→            {unreadCount > 0 && <span className="text-[10px] text-muted-foreground">{unreadCount} non lue(s)</span>}
   317→            <Button variant="ghost" size="sm" className="h-7 text-xs" onClick={markAllRead} disabled={unreadCount === 0}>Tout marquer lu</Button>
   318→          </div>
   319→        </div>
   320→        <div className="flex-1 overflow-y-auto">
   321→          {loading ? (
   322→            <div className="p-4 space-y-3">{Array.from({ length: 4 }).map((_, i) => <div key={i} className="flex gap-3"><Skeleton className="w-8 h-8 rounded-full" /><div className="flex-1 space-y-1"><Skeleton className="h-3 w-3/4" /><Skeleton className="h-2.5 w-1/2" /></div></div>)}</div>
   323→          ) : notifications.length === 0 ? (
   324→            <div className="p-8 text-center text-muted-foreground">
   325→              <Bell className="w-8 h-8 mx-auto mb-2 opacity-30" />
   326→              <p className="text-sm">Aucune notification</p>
   327→            </div>
   328→          ) : (
   329→            notifications.map(n => (
   330→              <div key={n.id} className={`flex items-start gap-3 p-4 hover:bg-muted/50 cursor-pointer transition-colors border-b last:border-0 ${!n.read ? 'bg-primary/5' : ''}`} onClick={() => { if (!n.read) markAsRead(n.id); }}>
   331→                <div className={`w-8 h-8 rounded-full flex items-center justify-center shrink-0 ${!n.read ? 'bg-primary/10 text-primary' : 'bg-muted text-muted-foreground'}`}>{notifIcons[n.type] || <Bell className="w-4 h-4" />}</div>
   332→                <div className="flex-1 min-w-0">
   333→                  <div className="flex items-center gap-2">
   334→                    <p className={`text-sm ${!n.read ? 'font-semibold' : 'font-medium'}`}>{n.title}</p>
   335→                    {!n.read && <div className="w-2 h-2 rounded-full bg-primary shrink-0" />}
   336→                  </div>
   337→                  <p className="text-xs text-muted-foreground mt-0.5 line-clamp-2">{n.message}</p>
   338→                  <p className="text-[10px] text-muted-foreground mt-1">{timeAgo(n.createdAt)}</p>
   339→                </div>
   340→              </div>
   341→            ))
   342→          )}
   343→        </div>
   344→      </div>
   345→    </>
   346→  );
   347→}
